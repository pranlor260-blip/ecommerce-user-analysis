-- 01_explore.sql  【数据探查SQL】

-- ① 每张表行数
SELECT 'users' AS 表名, COUNT(*) AS 行数 FROM users
UNION ALL
SELECT 'products', COUNT(*) FROM products
UNION ALL
SELECT 'orders', COUNT(*) FROM orders
UNION ALL
SELECT 'daily_targets', COUNT(*) FROM daily_targets;

-- ② 渠道
SELECT DISTINCT channel FROM orders;

-- ③ 城市
SELECT DISTINCT city FROM users ORDER BY city;

-- ④ 订单状态
SELECT DISTINCT order_status FROM orders;

-- ⑤ 日期范围
SELECT MIN(order_date), MAX(order_date) FROM orders;

-- ⑥ 缺失值
SELECT 
  SUM(CASE WHEN user_id IS NULL THEN 1 ELSE 0 END) AS user_id空,
  SUM(CASE WHEN product_id IS NULL THEN 1 ELSE 0 END) AS product_id空,
  SUM(CASE WHEN amount IS NULL THEN 1 ELSE 0 END) AS amount空,
  SUM(CASE WHEN channel IS NULL THEN 1 ELSE 0 END) AS channel空
FROM orders;

-- ⑦ 状态分布
SELECT order_status, COUNT(*) AS 数量
FROM orders
GROUP BY order_status;
